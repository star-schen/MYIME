cmake_minimum_required(VERSION 3.25)
project(MYIME VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The first vertical slice requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/EHsc>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set(RUST_PROFILE "debug" CACHE STRING "Cargo output profile")
add_executable(myime-probe tools/probe/main.cpp)
target_include_directories(myime-probe PRIVATE include)
target_link_libraries(myime-probe PRIVATE "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll.lib")
add_custom_target(stage_runtime ALL
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/$<CONFIG>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll" "${CMAKE_BINARY_DIR}/$<CONFIG>/myime_core.dll"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/.deps/librime/dist/lib/rime.dll" "${CMAKE_BINARY_DIR}/$<CONFIG>/rime.dll"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/target/maintenance-${RUST_PROFILE}/${RUST_PROFILE}/myime-tool.exe" "${CMAKE_BINARY_DIR}/$<CONFIG>/myime-tool.exe"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/themes" "${CMAKE_BINARY_DIR}/$<CONFIG>/themes")
add_library(myime_host SHARED windows/host/dll.cpp windows/host/service.cpp windows/host/candidate_window.cpp windows/host/myime.def windows/host/myime.rc)
target_link_libraries(myime_host PRIVATE ole32 oleaut32 advapi32 uuid user32 gdi32 shell32 userenv)
target_include_directories(myime_host PRIVATE include windows/host)
add_executable(myime-com-probe tools/probe/com.cpp)
target_include_directories(myime-com-probe PRIVATE windows/host)
target_link_libraries(myime-com-probe PRIVATE ole32 uuid)
add_dependencies(myime-com-probe myime_host)
add_executable(myime-tsf-probe tools/probe/tsf.cpp)
target_include_directories(myime-tsf-probe PRIVATE windows/host windows/compat)
target_link_libraries(myime-tsf-probe PRIVATE ole32 oleaut32 uuid user32 gdi32)
add_dependencies(myime-tsf-probe myime_host)
add_executable(myime-candidate-probe tools/probe/candidates.cpp)
target_include_directories(myime-candidate-probe PRIVATE windows/host)
target_link_libraries(myime-candidate-probe PRIVATE oleaut32 uuid)
add_executable(myime-compat windows/compat/main.cpp)
add_executable(myime-mode-probe tools/probe/mode.cpp windows/host/myime.rc)
target_include_directories(myime-mode-probe PRIVATE windows/host)
target_link_libraries(myime-mode-probe PRIVATE oleaut32 uuid user32)
target_link_libraries(myime-compat PRIVATE ole32 oleaut32 uuid user32 gdi32 imm32 comctl32)
add_executable(myime-theme-preview WIN32 windows/theme_preview/main.cpp windows/host/candidate_window.cpp)
target_include_directories(myime-theme-preview PRIVATE include windows/host)
target_link_libraries(myime-theme-preview PRIVATE user32 gdi32 shell32 userenv advapi32 ole32 uuid)
target_link_options(myime-theme-preview PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/windows/theme_preview/preview.manifest")
add_dependencies(myime-theme-preview stage_runtime)
add_executable(myime-theme-probe tools/probe/themes.cpp windows/host/candidate_window.cpp)
target_include_directories(myime-theme-probe PRIVATE include windows/host)
target_link_libraries(myime-theme-probe PRIVATE user32 gdi32 gdiplus "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll.lib")
add_dependencies(myime-theme-probe stage_runtime)
add_executable(myime-data-tool windows/data_tool/main.cpp)
target_include_directories(myime-data-tool PRIVATE include windows/host)
target_link_libraries(myime-data-tool PRIVATE ole32 "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll.lib")
add_dependencies(myime-data-tool stage_runtime)
add_executable(myime-workspace-probe tools/probe/workspace.cpp)
target_include_directories(myime-workspace-probe PRIVATE include windows/host)
target_link_libraries(myime-workspace-probe PRIVATE "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll.lib")
add_dependencies(myime-workspace-probe stage_runtime)
