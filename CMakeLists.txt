cmake_minimum_required(VERSION 3.25)
project(MYIME VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The first vertical slice requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/W4 /utf-8 /EHsc)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set(RUST_PROFILE "debug" CACHE STRING "Cargo output profile")
add_executable(myime-probe tools/probe/main.cpp)
target_include_directories(myime-probe PRIVATE include)
target_link_libraries(myime-probe PRIVATE "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll.lib")
add_custom_target(stage_runtime ALL
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/$<CONFIG>"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/target/${RUST_PROFILE}/myime_core.dll" "${CMAKE_BINARY_DIR}/$<CONFIG>/myime_core.dll"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/.deps/librime/dist/lib/rime.dll" "${CMAKE_BINARY_DIR}/$<CONFIG>/rime.dll")
add_library(myime_host SHARED windows/host/dll.cpp windows/host/service.cpp windows/host/candidate_window.cpp windows/host/myime.def)
target_link_libraries(myime_host PRIVATE ole32 oleaut32 advapi32 uuid user32 gdi32)
target_include_directories(myime_host PRIVATE include)
add_executable(myime-com-probe tools/probe/com.cpp)
target_include_directories(myime-com-probe PRIVATE windows/host)
target_link_libraries(myime-com-probe PRIVATE ole32 uuid)
add_dependencies(myime-com-probe myime_host)
add_executable(myime-tsf-probe tools/probe/tsf.cpp)
target_include_directories(myime-tsf-probe PRIVATE windows/host windows/compat)
target_link_libraries(myime-tsf-probe PRIVATE ole32 oleaut32 uuid user32 gdi32)
add_dependencies(myime-tsf-probe myime_host)
add_executable(myime-compat windows/compat/main.cpp)
target_link_libraries(myime-compat PRIVATE ole32 oleaut32 uuid user32 gdi32 imm32 comctl32)
